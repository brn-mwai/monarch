\section{The physics}
\act{04}{The physics}{What a measured field is worth.}

\begin{frame}{Media as a field on a coupled population}
\vspace{0.9em}
Each person holds an opinion $s_i = \pm 1$, is pulled toward their neighbours with strength
$J$, and is pushed by media with field $h$. Mean-field self-consistency:
\[
m = \tanh(\beta J m + h), \qquad h = \alpha X
\]
where $X$ is the measured observable and $\alpha$ is the unknown coupling.

\vspace{0.9em}
Expanding $\operatorname{artanh}(m) \approx m + m^3/3$ gives the Landau free energy
\[
F(m) = a m^2 + b m^4 - hm, \qquad a = \frac{1-\beta J}{2}, \qquad b = \frac{1}{12}
\]

\vspace{0.7em}
{\color{slate} Coefficients \textbf{derived} from the self-consistency condition. The
proposal carried $a = 1-\beta J$, $b = (\beta J)^3/3$, which do not follow from it. The
solver is checked against the exact exponents $\beta = 1/2$, $\gamma = 1$, $\delta = 3$.}
\end{frame}

\begin{frame}{Fitting the coupling fails, and that is informative}
\vspace{1.0em}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{What was tried.} Regress the outcome on the measured index and read off $\alpha$.

\vspace{0.8em}
\textbf{What happened.} The fit performs \emph{worse than the sample mean} out of sample, and
the estimate scales inversely with a coupling $\beta J$ that no measurement here fixes.

\vspace{0.8em}
\alert{So no value of $\hat{\alpha}$ is quoted. Anywhere.} Not in a table, not in a caption,
not for illustration.
\end{column}
\begin{column}{0.44\textwidth}
\vspace{0.4em}
{\color{slate}\small Reporting an unidentified coupling as though it were a measured one
would be the failure here. Declining to quote it is not.}

\vspace{1.2em}
{\color{slate}\small What replaces it is a statement that does not need the fit at all.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The bound: what any content observable must satisfy}
\vspace{0.3em}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\vspace{0.9em}
\begin{center}
{\fontsize{24}{28}\selectfont $\alpha \;\ge\; \dfrac{h_c(\beta J)}{\Delta X}$}
\end{center}

\vspace{1.0em}
{\small No transition is possible unless the coupling clears this. Rather than fitting
$\alpha$ and hoping it clears zero, it states what $\alpha$ \emph{would have to be} for the
mechanism to work at all --- for \textbf{any} candidate observable, before a corpus exists.}
\end{column}
\begin{column}{0.50\textwidth}
\includegraphics[width=\textwidth]{F6_alpha_required.pdf}
\end{column}
\end{columns}

\vspace{0.5em}
\begin{center}
\small Measured spread $\Delta X = 0.1241$ \;$\Rightarrow$\;
$\alpha \ge 0.169$ at $\beta J = 1.10$, \; $1.655$ at $1.50$, \; $\mathbf{4.294}$ at $2.00$
\end{center}
\end{frame}
