\section{The physics}
\act{04}{The physics}{What a measured field is worth.}

\begin{frame}{Media enters the mean-field model as a field on the measured observable}
\vspace{0.7em}
Each person holds an opinion $s_i = \pm 1$, is pulled toward their neighbours with strength
$J$, and is pushed by media with field $h$:
\[
m = \tanh(\beta J m + h), \qquad h = \alpha X
\]
where $X$ is the measured observable and $\alpha$ the unknown coupling. Expanding
$\operatorname{artanh}(m) \approx m + m^3/3$:
\[
F(m) = a m^2 + b m^4 - hm, \qquad a = \frac{1-\beta J}{2}, \qquad b = \frac{1}{12}
\]

\vspace{0.4em}
{\color{slate} Coefficients \textbf{derived} from the self-consistency condition. The proposal
carried $a = 1-\beta J$, $b = (\beta J)^3/3$, which do not follow from it. The solver
reproduces the exact exponents $\beta = 1/2$, $\gamma = 1$, $\delta = 3$.}
\end{frame}

\begin{frame}{The corpus does not identify the coupling, so no value is quoted}
\vspace{1.0em}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{What was tried.} Regress the outcome on the measured index and read off $\alpha$.

\vspace{0.8em}
\textbf{What happened.} The fit performs \emph{worse than the sample mean} out of sample, and
the estimate scales inversely with a coupling $\beta J$ that no measurement here fixes.

\vspace{0.8em}
\alert{So no value of $\hat{\alpha}$ appears anywhere} --- not in a table, a caption, or
``for illustration''.
\end{column}
\begin{column}{0.44\textwidth}
\vspace{0.3em}
{\color{slate}\small Reporting an unidentified coupling as though it were measured would be
the failure here. Declining to quote it is not.

\vspace{1.0em}
What replaces it is a statement that does not need the fit at all.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Any content observable must clear a bound to drive a transition}
\vspace{0.2em}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\includegraphics[width=\textwidth]{F6_alpha_required.pdf}
\end{column}
\begin{column}{0.46\textwidth}
\vspace{0.5em}
\begin{center}
{\fontsize{23}{27}\selectfont\color{navy}
$\alpha \;\ge\; \dfrac{h_c(\beta J)}{\Delta X}$}
\end{center}

\vspace{0.9em}
{\small Rather than fitting $\alpha$ and hoping it clears zero, this states what $\alpha$
\emph{would have to be} for the mechanism to work at all --- for \textbf{any} candidate
observable, before a corpus exists.}
\end{column}
\end{columns}

\vspace{0.4em}
\begin{center}
\small Measured spread $\Delta X = 0.1241$ \;$\Rightarrow$\;
$\alpha \ge 0.169$ at $\beta J = 1.10$, \; $1.655$ at $1.50$, \; $\mathbf{4.294}$ at $2.00$
\end{center}
\end{frame}
