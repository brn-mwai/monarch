% The submission document, set for two-sided binding as a booklet.
% build_check.tex stays as the fast chapter-only compile and shares the chapters verbatim.
\documentclass[12pt,a4paper,twoside,openright]{report}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{mathpazo}
\usepackage{microtype}

% Binding edge takes the wider margin. A bound booklet loses 10 to 15 mm into the gutter, so
% inner exceeds outer and the two-sided layout alternates which physical edge that is.
\usepackage[a4paper,inner=3.5cm,outer=2.5cm,top=3cm,bottom=3cm,headsep=1cm]{geometry}
\usepackage{setspace}
\usepackage{amsmath}
\usepackage{booktabs}
\usepackage{enumitem}
\usepackage{graphicx}
\usepackage{array}
\usepackage{emptypage}
\usepackage[font=small,labelfont=bf,width=0.9\textwidth]{caption}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage[hidelinks]{hyperref}

\setstretch{1.5}

% Chapter openings: number on its own line above a rule, so a reader flicking through a bound
% copy lands on chapter starts.
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries}
  {\normalsize\scshape\chaptertitlename\ \thechapter}
  {8pt}
  {\Huge}
  [\vspace{4pt}\titlerule]
\titlespacing*{\chapter}{0pt}{10pt}{28pt}

\titleformat{\section}{\normalsize\bfseries\scshape}{\thesection}{0.6em}{}
\titleformat{\subsection}{\normalsize\bfseries}{\thesubsection}{0.6em}{}

\pagestyle{fancy}
\fancyhf{}
% Verso carries the chapter, recto the section, which is the convention that lets someone
% holding an open booklet know where they are from either page.
\fancyhead[LE]{\footnotesize\nouppercase{\leftmark}}
\fancyhead[RO]{\footnotesize\nouppercase{\rightmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\chaptermark}[1]{\markboth{\thechapter.\ #1}{}}
\renewcommand{\sectionmark}[1]{\markright{\thesection\ #1}}

\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% A single short line stranded at a page break reads as a printing fault in a bound copy.
\widowpenalty=10000
\clubpenalty=10000
\setlength{\emergencystretch}{2em}

\begin{document}

\input{frontmatter.tex}

\input{ch1_introduction.tex}
\input{ch2_literature.tex}
\input{ch3_framework.tex}
\input{ch4_methodology.tex}
\input{ch5_results.tex}
\input{ch6_discussion.tex}
\input{ch7_conclusion.tex}

\cleardoublepage
\input{bibliography.tex}

\end{document}
